function savePlayer(data) {
  localStorage.setItem("novas_player", JSON.stringify(data));
}

function getXPForLevel(level) {
  return Math.floor(100 * Math.pow(1.5, level - 1));
}

function getRank(score) {
  let rank = RANKS[0];
  for (const r of RANKS) {
    if (score >= r.min) rank = r;
  }
  return rank;
}

function AudioEngine() {
  const ctx = useRef(null);
  const gainNode = useRef(null);

  const init = useCallback(() => {
    if (!ctx.current) {
      ctx.current = new (window.AudioContext || window.webkitAudioContext)();
      gainNode.current = ctx.current.createGain();
      gainNode.current.connect(ctx.current.destination);
    }
  }, []);

  const play = useCallback((type, volume = 0.3) => {
    try {
      init();
      if (ctx.current.state === "suspended") ctx.current.resume();
      const osc = ctx.current.createOscillator();
      const gain = ctx.current.createGain();
      osc.connect(gain);
      gain.connect(gainNode.current);

      const configs = {
        shoot: { freq: 800, type: "square", duration: 0.08, attack: 0, decay: 0.08 },
        hit: { freq: 200, type: "sawtooth", duration: 0.15, attack: 0, decay: 0.15 },
        explosion: { freq: 80, type: "sawtooth", duration: 0.3, attack: 0, decay: 0.3 },
        powerup: { freq: 600, type: "sine", duration: 0.4, attack: 0.05, decay: 0.35 },
        levelup: { freq: 440, type: "sine", duration: 0.6, attack: 0.05, decay: 0.55 },
        gameover: { freq: 150, type: "sawtooth", duration: 0.8, attack: 0, decay: 0.8 },
        click: { freq: 1000, type: "square", duration: 0.05, attack: 0, decay: 0.05 },
        coin: { freq: 1200, type: "sine", duration: 0.2, attack: 0, decay: 0.2 },
        achievement: { freq: 880, type: "sine", duration: 0.5, attack: 0.02, decay: 0.48 },
      };

      const c = configs[type] || configs.click;
      osc.type = c.type;
      osc.frequency.setValueAtTime(c.freq, ctx.current.currentTime);
      if (type === "explosion") {
        osc.frequency.exponentialRampToValueAtTime(20, ctx.current.currentTime + c.duration);
      } else if (type === "levelup") {
        osc.frequency.setValueAtTime(440, ctx.current.currentTime);
        osc.frequency.setValueAtTime(550, ctx.current.currentTime + 0.15);
        osc.frequency.setValueAtTime(660, ctx.current.currentTime + 0.3);
      }
      gain.gain.setValueAtTime(0, ctx.current.currentTime);
      gain.gain.linearRampToValueAtTime(volume, ctx.current.currentTime + c.attack + 0.01);
      gain.gain.exponentialRampToValueAtTime(0.001, ctx.current.currentTime + c.duration);
      osc.start(ctx.current.currentTime);
      osc.stop(ctx.current.currentTime + c.duration + 0.01);
    } catch (e) {}
  }, [init]);

  return { play };
}

  const audio = AudioEngine();
  const splashTimer = useRef(null);

  useEffect(() => {
    savePlayer(player);
  }, [player]);

  useEffect(() => {
    splashTimer.current = setTimeout(() => setScreen(SCREENS.DASHBOARD), 2500);
    return () => clearTimeout(splashTimer.current);
  }, []);

  const showNotif = useCallback((msg, type = "info", icon = "ℹ️") => {
    setNotification({ msg, type, icon, id: Date.now() });
    setTimeout(() => setNotification(null), 3000);
  }, []);

  const t = LANGUAGES[player.settings.language] || LANGUAGES.fr;
  const theme = THEMES[player.theme] || THEMES.default;

  const updatePlayer = useCallback((updates) => {
    setPlayer(prev => {
      const updated = typeof updates === "function" ? updates(prev) : { ...prev, ...updates };
      savePlayer(updated);
      return updated;
    });
  }, []);

  const buyItem = useCallback((item) => {
    if (item.real) {
      showNotif("Paiement requis - intégration Stripe en production", "info", "💳");
      return;
    }
    if (player.coins < item.price) {
      showNotif("Coins insuffisants!", "error", "❌");
      audio.play("hit");
      return;
    }
    if (player.inventory.includes(item.id)) {
      showNotif("Déjà possédé!", "info", "ℹ️");
      return;
    }
    updatePlayer(prev => ({
      ...prev,
      coins: prev.coins - item.price,
      inventory: [...prev.inventory, item.id],
    }));
    showNotif(`${item.name} acheté!`, "success", "🎉");
    audio.play("coin");
  }, [player, updatePlayer, showNotif, audio]);

  const unlockAchievement = useCallback((id) => {
    if (player.achievements.includes(id)) return;
    const ach = ACHIEVEMENTS_DATA.find(a => a.id === id);
    if (!ach) return;
    updatePlayer(prev => ({
      ...prev,
      achievements: [...prev.achievements, id],
      coins: prev.coins + ach.points,
    }));
    showNotif(`🏆 Succès: ${ach.title}! +${ach.points} coins`, "achievement", "🏆");
    audio.play("achievement");
  }, [player.achievements, updatePlayer, showNotif, audio]);

  const onGameEnd = useCallback((result) => {
    const { score, kills, survived, combo, level: lvl, perfect, timeMs } = result;
    updatePlayer(prev => {
      let newXP = prev.xp + Math.floor(score / 10) + kills * 5;
      let newLevel = prev.level;
      while (newXP >= getXPForLevel(newLevel)) {
        newXP -= getXPForLevel(newLevel);
        newLevel++;
      }
      const gained = newLevel > prev.level;
      const newCoins = prev.coins + Math.floor(score / 100) + kills * 2;
      const newAchs = [...prev.achievements];

      const check = (id, cond) => { if (cond && !newAchs.includes(id)) { newAchs.push(id); setTimeout(() => showNotif(`🏆 ${ACHIEVEMENTS_DATA.find(a=>a.id===id)?.title}!`, "achievement", "🏆"), 1500); audio.play("achievement"); } };
      check("first_kill", kills > 0);
      check("combo_5", combo >= 5);
      check("combo_10", combo >= 10);
      check("score_1000", score >= 1000);
      check("score_10000", score >= 10000);
      check("score_50000", score >= 50000);
      check("level_5", newLevel >= 5);
      check("level_10", newLevel >= 10);
      check("survive_120", survived >= 120);
      check("perfect_level", perfect);
      check("play_10", prev.totalGames + 1 >= 10);
      check("play_50", prev.totalGames + 1 >= 50);
      check("kill_100", prev.totalKills + kills >= 100);
      check("kill_500", prev.totalKills + kills >= 500);
      check("speed_demon", timeMs < 30000 && lvl >= 1);

      if (gained) { setTimeout(() => { showNotif(`🎉 Niveau ${newLevel}!`, "levelup", "🚀"); audio.play("levelup"); }, 500); }

      return {
        ...prev,
        xp: newXP,
        level: newLevel,
        coins: newCoins,
        totalScore: prev.totalScore + score,
        bestScore: Math.max(prev.bestScore, score),
        totalKills: prev.totalKills + kills,
        totalGames: prev.totalGames + 1,
        bestCombo: Math.max(prev.bestCombo, combo),
        totalTimePlayed: prev.totalTimePlayed + Math.floor(survived),
        achievements: newAchs,
      };
    });
    if (score > 0) setScreen(SCREENS.GAME_OVER);
  }, [updatePlayer, showNotif, audio]);

  const rank = getRank(player.bestScore);
  const xpNeeded = getXPForLevel(player.level);
  const xpPct = Math.min(100, (player.xp / xpNeeded) * 100);

  return (
    <div style={{
      width: "100%",
      minHeight: "100vh",
      background: theme.bg,
      color: theme.text,
      fontFamily: "'Courier New', monospace",
      overflow: "hidden",
      position: "relative",
    }}>
      {/* NOTIFICATION */}
      {notification && (
        <div style={{
          position: "fixed",
          top: 20,
          left: "50%",
          transform: "translateX(-50%)",
          background: notification.type === "error" ? "#7F1D1D" : notification.type === "achievement" ? "#78350F" : notification.type === "success" ? "#064E3B" : "#1E3A5F",
          border: `1px solid ${notification.type === "error" ? "#EF4444" : notification.type === "achievement" ? "#F59E0B" : notification.type === "success" ? "#10B981" : theme.accent}`,
          color: "#FFFFFF",
          padding: "12px 24px",

          boxShadow: "0 4px 20px rgba(0,0,0,0.5)",

        }}>
          {notification.icon} {notification.msg}

      <style>{`

function GameCanvas({ player, gameMode, theme, audio, onEnd, onQuit }) {
  const canvasRef = useRef(null);
  const stateRef = useRef(null);
  const animRef = useRef(null);
  const keysRef = useRef({});
  const touchRef = useRef({});
  const lastShotRef = useRef(0);
  const lastEnemyShotRef = useRef({});

  useEffect(() => {
    const canvas = canvasRef.current;
    const ctx = canvas.getContext("2d");
    const W = canvas.width = window.innerWidth > 480 ? 480 : window.innerWidth;
    const H = canvas.height = window.innerHeight;

    const ENEMY_COLS = 6, ENEMY_ROWS = 3;
    const isHardcore = gameMode === GAME_MODES.HARDCORE;
    const isSurvival = gameMode === GAME_MODES.SURVIVAL;
    const isTimeAttack = gameMode === GAME_MODES.TIME_ATTACK;
    const isZen = gameMode === GAME_MODES.ZEN;
    const isBossRush = gameMode === GAME_MODES.BOSS_RUSH;

    function makeStars() {
      return Array.from({length:100}, () => ({ x:Math.random()*W, y:Math.random()*H, speed:Math.random()*0.5+0.1, size:Math.random()*1.5+0.5, brightness:Math.random() }));
    }

    function makeEnemies(lvl) {
      const enemies = [];
      const types = ["basic","scout","heavy","zigzag","sniper"];
      for (let r=0; r<ENEMY_ROWS; r++) {
        for (let c=0; c<ENEMY_COLS; c++) {
          const typeIdx = Math.min(Math.floor(lvl/3)+r, types.length-1);
          const type = types[typeIdx];
          const hp = type==="heavy"?3:type==="sniper"?2:1;
          enemies.push({
            x: 40 + c*(W-80)/ENEMY_COLS + (W-80)/(ENEMY_COLS*2),
            y: 80 + r*55,
            w:32, h:24, alive:true, type,
            hp, maxHp:hp,
            hitFlash:0, deathAnim:0,
            id:`${r}-${c}`,
          });
        }
      }
      return enemies;
    }

    function makeBoss(lvl) {
      return {
        x: W/2, y:100,
        w:64, h:48,
        hp: 15+lvl*5, maxHp:15+lvl*5,
        alive:true, isBoss:true,
        dir:1, speed:1.5+lvl*0.3,
        phase:1, hitFlash:0, angle:0,
        shieldActive: lvl >= 3,
        shieldHp: lvl >= 3 ? 5 : 0,
      };
    }

    function spawnPowerup(x, y) {
      if (Math.random()<0.15) {
        const types = ["shield","rapid","triple","bomb","slow"];
        return { x, y, type:types[Math.floor(Math.random()*types.length)], vy:1.5, active:true, pulse:0 };
      }
      return null;
    }

    const startLives = isHardcore ? 1 : isZen ? 999 : 3;

    stateRef.current = {
      player: { x:W/2-16, y:H-90, w:32, h:32, lives:startLives, shield:0, rapid:0, triple:0, hit:0 },
      bullets: [],
      enemyBullets: [],
      enemies: isBossRush ? [] : makeEnemies(0),
      boss: isBossRush ? makeBoss(0) : null,
      bossLevel: 0,
      particles: [],
      powerups: [],
      stars: makeStars(),
      score: 0,
      combo: 0,
      comboTimer: 0,
      maxCombo: 0,
      level: 1,
      enemyDir: 1,
      enemyDrop: false,
      gameOver: false,
      victory: false,
      paused: false,
      startTime: Date.now(),
      elapsed: 0,
      perfect: true,
      scoreMulti: 1,
      shakeTimer: 0,
      shakeX: 0, shakeY: 0,
      flashTimer: 0,
      killCount: 0,
      timeLimit: isTimeAttack ? 60000 : Infinity,
    };

    function addParticle(x, y, color="#3B82F6", count=8, speed=3, size=3, life=40) {
      if (player.settings.particleDensity === "low" && count > 5) count = 3;
      for (let i=0; i<count; i++) {
        const angle = (Math.PI*2/count)*i + Math.random()*0.5;
        stateRef.current.particles.push({ x, y, vx:Math.cos(angle)*(speed+Math.random()*speed), vy:Math.sin(angle)*(speed+Math.random()*speed), color, life, maxLife:life, size:size+Math.random()*size });
      }
    }

    function addTextParticle(x, y, text, color="#FFFFFF") {
      stateRef.current.particles.push({ x, y, text, color, vy:-1.5, life:60, maxLife:60, isText:true });
    }

    let frameCount = 0;
    let lastFPSTime = Date.now();
    let fps = 60;

    function gameLoop() {
      const s = stateRef.current;
      if (!s || s.paused) { animRef.current = requestAnimationFrame(gameLoop); return; }

      frameCount++;
      if (frameCount % 30 === 0) {
        const now = Date.now();
        fps = Math.round(30000/(now-lastFPSTime));
        lastFPSTime = now;
      }

      s.elapsed = (Date.now()-s.startTime)/1000;
      if (isTimeAttack && Date.now()-s.startTime >= s.timeLimit) {
        s.victory = true;
        endGame();
        return;
      }

      // Screen shake
      if (s.shakeTimer > 0) {
        s.shakeTimer--;
        s.shakeX = (Math.random()-0.5)*6;
        s.shakeY = (Math.random()-0.5)*6;
      } else { s.shakeX=0; s.shakeY=0; }

      ctx.save();
      if (player.settings.screenShake && (s.shakeX||s.shakeY)) ctx.translate(s.shakeX, s.shakeY);

      // Background
      ctx.fillStyle = "#000000";
      ctx.fillRect(0,0,W,H);

      // Stars
      for (const star of s.stars) {
        star.y += star.speed;
        if (star.y > H) { star.y=0; star.x=Math.random()*W; }
        ctx.globalAlpha = 0.3+star.brightness*0.7;
        ctx.fillStyle = "#FFFFFF";
        ctx.beginPath();
        ctx.arc(star.x, star.y, star.size, 0, Math.PI*2);
        ctx.fill();
      }
      ctx.globalAlpha = 1;

      // Player movement
      const p = s.player;
      const spd = 4;
      if ((keysRef.current["ArrowUp"]||keysRef.current["w"]||touchRef.current.up) && p.y > 50) p.y -= spd;
      if ((keysRef.current["ArrowDown"]||keysRef.current["s"]||touchRef.current.down) && p.y < H-60) p.y += spd;
      if ((keysRef.current["ArrowLeft"]||keysRef.current["a"]||touchRef.current.left) && p.x > 0) p.x -= spd;
      if ((keysRef.current["ArrowRight"]||keysRef.current["d"]||touchRef.current.right) && p.x < W-p.w) p.x += spd;

      // Auto-fire
      const now = Date.now();
      const fireRate = p.rapid > 0 ? 120 : 300;
      if (now - lastShotRef.current > fireRate) {
        lastShotRef.current = now;
        const bulletColor = player.equippedBullet==="bullet_plasma"?"#7C3AED":player.equippedBullet==="bullet_laser"?"#DC2626":"#60A5FA";
        s.bullets.push({ x:p.x+p.w/2-2, y:p.y, w:4, h:12, vy:-10, color:bulletColor });
        if (p.triple > 0) {
          s.bullets.push({ x:p.x+p.w/2-14, y:p.y+8, w:4, h:10, vy:-9, vx:-2, color:bulletColor });
          s.bullets.push({ x:p.x+p.w/2+10, y:p.y+8, w:4, h:10, vy:-9, vx:2, color:bulletColor });
        }
        audio.play("shoot", 0.1);
      }

      // Update bullets
      s.bullets = s.bullets.filter(b => {
        b.x += b.vx||0;
        b.y += b.vy;
        if (b.y < -20 || b.x < -20 || b.x > W+20) return false;

        // Hit boss
        if (s.boss && s.boss.alive) {
          const bs = s.boss;
          if (b.x>bs.x-bs.w/2&&b.x<bs.x+bs.w/2&&b.y>bs.y-bs.h/2&&b.y<bs.y+bs.h/2) {
            if (bs.shieldHp > 0) {
              bs.shieldHp--;
              bs.hitFlash=8;
              addParticle(b.x,b.y,"#60A5FA",5,4,2,20);
            } else {
              bs.hp--;
              bs.hitFlash=8;
              addParticle(b.x,b.y,"#EF4444",8,5,3,30);
              addTextParticle(b.x,b.y-10,"-1","#EF4444");
              s.score += 50;
              if (bs.hp <= 0) {
                bs.alive = false;
                addParticle(bs.x,bs.y,"#F59E0B",40,8,5,80);
                addTextParticle(bs.x,bs.y-20,"BOSS VAINCU! +1000","#F59E0B");
                s.score += 1000;
                s.killCount++;
                audio.play("explosion",0.4);
                if (isBossRush) { s.bossLevel++; setTimeout(()=>{ s.boss=makeBoss(s.bossLevel); },2000); }
                else { s.victory=true; endGame(); return false; }
              }
            }
            return false;
          }
        }

        // Hit enemies
        for (const e of s.enemies) {
          if (!e.alive) continue;
          if (b.x>e.x-e.w/2&&b.x<e.x+e.w/2&&b.y>e.y-e.h/2&&b.y<e.y+e.h/2) {
            e.hp--;
            e.hitFlash = 8;
            audio.play("hit",0.15);
            if (e.hp <= 0) {
              e.alive = false;
              e.deathAnim = 1;
              const pts = (e.type==="heavy"?30:e.type==="sniper"?25:e.type==="scout"?15:e.type==="zigzag"?20:10) * s.scoreMulti;
              s.score += pts;
              s.killCount++;
              s.combo++;
              s.comboTimer = 120;
              if (s.combo > s.maxCombo) s.maxCombo = s.combo;
              const comboBonus = Math.min(s.combo, 10);
              s.scoreMulti = 1 + comboBonus * 0.1;
              addParticle(e.x, e.y, e.type==="heavy"?"#EF4444":e.type==="sniper"?"#7C3AED":"#F59E0B", 12, 5, 3, 50);
              addTextParticle(e.x, e.y-15, `+${pts}`, "#F59E0B");
              if (s.combo > 1) addTextParticle(e.x, e.y-30, `x${s.combo} COMBO!`, "#FF6B6B");
              const pu = spawnPowerup(e.x, e.y);
              if (pu) s.powerups.push(pu);
              audio.play("explosion",0.2);
            } else {
              addParticle(e.x, e.y, "#60A5FA", 4, 3, 2, 20);
              addTextParticle(e.x, e.y-10, "-1", "#60A5FA");
            }
            return false;
          }
        }
        return true;
      });

      // Combo timer
      if (s.comboTimer > 0) {
        s.comboTimer--;
        if (s.comboTimer === 0) { s.combo=0; s.scoreMulti=1; }
      }

      // Enemy movement & logic
      if (!s.boss) {
        const alive = s.enemies.filter(e=>e.alive);
        if (alive.length === 0 && !isSurvival && !isZen) {
          s.level++;
          if (s.level % 5 === 0 && !isBossRush) {
            s.boss = makeBoss(s.level/5);
            addTextParticle(W/2,H/2,"⚠️ BOSS !", "#EF4444");
          } else {
            s.enemies = makeEnemies(s.level-1);
            addTextParticle(W/2,H/2,`NIVEAU ${s.level}!`,"#10B981");
            audio.play("levelup",0.3);
          }
        }

        // Survival - spawn enemies
        if ((isSurvival||isZen) && Math.random()<0.008+s.elapsed*0.0002) {
          const typeRng = Math.random();
          const type = typeRng<0.4?"basic":typeRng<0.7?"scout":typeRng<0.85?"heavy":"zigzag";
          s.enemies.push({ x:40+Math.random()*(W-80), y:-30, w:32, h:24, alive:true, type, hp:1, maxHp:1, hitFlash:0, deathAnim:0, id:`sp${Date.now()}${Math.random()}`, speedY:1+s.elapsed*0.01 });
        }

        let maxX=0, minX=W;
        for (const e of alive) { if (e.x>maxX) maxX=e.x; if (e.x<minX) minX=e.x; }
        const speed = (0.5+s.level*0.1) * (1+Math.max(0,(ENEMY_COLS*ENEMY_ROWS-alive.length)*0.05));
        s.enemyDrop = (maxX+20>W-10&&s.enemyDir>0)||(minX-20<10&&s.enemyDir<0);
        if (s.enemyDrop) s.enemyDir *= -1;

        for (const e of s.enemies) {
          e.hitFlash = Math.max(0,e.hitFlash-1);
          if (!e.alive) continue;
          if (isSurvival||isZen) {
            e.y += e.speedY||1;
            e.x += Math.sin(e.y*0.05)*1.5;
            if (e.y > H+40) { e.alive=false; }
          } else {
            e.x += speed*s.enemyDir;
            if (s.enemyDrop) e.y += 20;
            if (e.type==="zigzag") e.x += Math.sin(Date.now()*0.003+e.y)*1.5;
          }
        }

        // Enemy shooting (max 5 bullets per 2s per enemy)
        const shotInterval = 2000;
        for (const e of alive) {
          if (!e.alive) continue;
          if (e.type==="sniper"||e.type==="heavy"||(Math.random()<0.002)) {
            const eid = e.id;
            const lastShot = lastEnemyShotRef.current[eid]||0;
            if (now-lastShot > shotInterval) {
              lastEnemyShotRef.current[eid] = now;
              const angle = e.type==="sniper" ? Math.atan2(p.y-e.y, p.x+p.w/2-e.x) : Math.PI/2;
              s.enemyBullets.push({ x:e.x, y:e.y+e.h/2, w:3, h:8, vx:Math.cos(angle)*4, vy:Math.sin(angle)*4+3, color:e.type==="heavy"?"#EF4444":"#F97316" });
            }
          }
        }
      }

      // Boss logic
      if (s.boss && s.boss.alive) {
        const bs = s.boss;
        bs.hitFlash = Math.max(0,bs.hitFlash-1);
        bs.angle += 0.02;
        bs.x += bs.dir*bs.speed;
        if (bs.x+bs.w/2 > W-20) bs.dir=-1;
        if (bs.x-bs.w/2 < 20) bs.dir=1;
        if (bs.hp < bs.maxHp*0.5) bs.phase=2;
        if (bs.hp < bs.maxHp*0.25) bs.phase=3;

        // Boss firing pattern - max 5 shots per 2s
        const bossId = "boss";
        const lastBossShot = lastEnemyShotRef.current[bossId]||0;
        const bossFireRate = 2000/(bs.phase);
        if (now-lastBossShot > bossFireRate) {
          lastEnemyShotRef.current[bossId] = now;
          const shots = Math.min(5, bs.phase===3?5:bs.phase===2?3:1);
          for (let i=0;i<shots;i++) {
            const a = bs.phase===1?Math.PI/2:bs.phase===2?(Math.PI/2+(i-1)*0.3):(Math.PI/4+i*0.3);
            s.enemyBullets.push({ x:bs.x, y:bs.y+bs.h/2, w:5, h:10, vx:Math.cos(a)*5, vy:Math.sin(a)*5, color:"#EF4444", isBoss:true });
          }
        }
      }

      // Enemy bullets
      s.enemyBullets = s.enemyBullets.filter(b => {
        b.x += b.vx||0;
        b.y += b.vy;
        if (b.y>H+20||b.x<-20||b.x>W+20) return false;
        const hit = b.x>p.x&&b.x<p.x+p.w&&b.y>p.y&&b.y<p.y+p.h;
        if (hit && !isZen) {
          if (p.shield>0) {
            p.shield=0;
            addParticle(p.x+p.w/2,p.y,"#60A5FA",10,5,3,30);
            addTextParticle(p.x+p.w/2,p.y-20,"BOUCLIER!","#60A5FA");
          } else {
            p.lives--;
            p.hit=30;
            s.perfect=false;
            s.shakeTimer=12;
            s.flashTimer=10;
            addParticle(p.x+p.w/2,p.y+p.h/2,"#EF4444",20,6,4,50);
            audio.play("hit",0.4);
            if (p.lives<=0 && !isZen) { s.gameOver=true; endGame(); return false; }
          }
          return false;
        }
        return true;
      });

      // Powerups
      s.powerups = s.powerups.filter(pu => {
        if (!pu.active) return false;
        pu.y += pu.vy;
        pu.pulse++;
        if (pu.y>H+30) return false;
        if (pu.x>p.x-20&&pu.x<p.x+p.w+20&&pu.y>p.y&&pu.y<p.y+p.h+30) {
          const dur=300;
          if (pu.type==="shield") p.shield=1;
          else if (pu.type==="rapid") p.rapid=dur;
          else if (pu.type==="triple") p.triple=dur;
          else if (pu.type==="bomb") { s.enemies.forEach(e=>{if(e.alive){e.alive=false;addParticle(e.x,e.y,"#F59E0B",10,5,3,40);s.score+=10;}}); }
          else if (pu.type==="slow") s.scoreMulti=Math.min(3,s.scoreMulti+0.5);
          addTextParticle(p.x+p.w/2,p.y-20,pu.type.toUpperCase()+"!","#10B981");
          audio.play("powerup",0.25);
          return false;
        }
        return true;
      });

      if (p.rapid>0) p.rapid--;
      if (p.triple>0) p.triple--;
      if (p.hit>0) p.hit--;

      // Flash overlay
      if (s.flashTimer>0) { s.flashTimer--; ctx.fillStyle=`rgba(255,0,0,${s.flashTimer*0.04})`; ctx.fillRect(0,0,W,H); }

      // Particles
      s.particles = s.particles.filter(pt => {
        pt.life--;
        if (pt.life<=0) return false;
        if (pt.isText) {
          pt.y += pt.vy;
          ctx.globalAlpha = pt.life/pt.maxLife;
          ctx.fillStyle = pt.color;
          ctx.font = "bold 12px 'Courier New'";
          ctx.textAlign = "center";
          ctx.fillText(pt.text, pt.x, pt.y);
          ctx.textAlign = "left";
        } else {
          pt.x += pt.vx; pt.y += pt.vy; pt.vx*=0.95; pt.vy*=0.95;
          ctx.globalAlpha = pt.life/pt.maxLife;
          ctx.fillStyle = pt.color;
          ctx.beginPath();
          ctx.arc(pt.x, pt.y, pt.size*(pt.life/pt.maxLife), 0, Math.PI*2);
          ctx.fill();
        }
        return true;
      });
      ctx.globalAlpha=1;

      // Draw powerups
      for (const pu of s.powerups) {
        const icons = {shield:"🛡️",rapid:"⚡",triple:"🔱",bomb:"💣",slow:"✨"};
        const colors = {shield:"#60A5FA",rapid:"#F59E0B",triple:"#A78BFA",bomb:"#EF4444",slow:"#10B981"};
        ctx.save();
        ctx.globalAlpha = 0.7+Math.sin(pu.pulse*0.1)*0.3;
        ctx.fillStyle = colors[pu.type]||"#fff";
        ctx.strokeStyle = colors[pu.type]||"#fff";
        ctx.lineWidth = 2;
        ctx.beginPath();
        ctx.arc(pu.x, pu.y, 14, 0, Math.PI*2);
        ctx.stroke();
        ctx.font = "16px serif";
        ctx.textAlign = "center";
        ctx.fillText(icons[pu.type]||"?", pu.x, pu.y+6);
        ctx.textAlign = "left";
        ctx.restore();
      }

      // Draw enemies
      for (const e of s.enemies) {
        if (!e.alive) continue;
        ctx.save();
        if (e.hitFlash>0) { ctx.globalAlpha = 0.5+Math.sin(e.hitFlash*1.5)*0.5; }
        const colors = {basic:"#10B981",scout:"#60A5FA",heavy:"#EF4444",zigzag:"#F59E0B",sniper:"#A78BFA"};
        const shapes = {basic:"▼",scout:"◆",heavy:"★",zigzag:"✦",sniper:"▲"};
        ctx.fillStyle = colors[e.type]||"#888";
        ctx.shadowColor = colors[e.type]||"#888";
        ctx.shadowBlur = 8;
        ctx.font = `${e.hitFlash>0?28:26}px serif`;
        ctx.textAlign = "center";
        ctx.fillText(shapes[e.type]||"▼", e.x, e.y+8);
        ctx.textAlign = "left";
        if (e.hp > 1) {
          const bw = e.w; const bx = e.x-bw/2; const by = e.y-e.h/2-8;
          ctx.fillStyle="#1F2937"; ctx.fillRect(bx,by,bw,4);
          ctx.fillStyle=colors[e.type]||"#fff"; ctx.fillRect(bx,by,bw*(e.hp/e.maxHp),4);
        }
        ctx.restore();
      }

      // Draw boss
      if (s.boss && s.boss.alive) {
        const bs = s.boss;
        ctx.save();
        ctx.translate(bs.x, bs.y);
        if (bs.hitFlash>0) ctx.globalAlpha=0.5+Math.sin(bs.hitFlash)*0.5;
        const bossColor = bs.phase===3?"#EF4444":bs.phase===2?"#F59E0B":"#A78BFA";
        ctx.strokeStyle=bossColor; ctx.lineWidth=2; ctx.shadowColor=bossColor; ctx.shadowBlur=15;
        ctx.beginPath();
        for (let i=0;i<8;i++) {
          const a=bs.angle+i*Math.PI/4;
          const r=bs.phase===3?30+Math.sin(bs.angle*5)*8:28;
          i===0?ctx.moveTo(Math.cos(a)*r,Math.sin(a)*r):ctx.lineTo(Math.cos(a)*r,Math.sin(a)*r);
        }
        ctx.closePath(); ctx.stroke();
        ctx.font="bold 24px serif"; ctx.textAlign="center"; ctx.fillStyle=bossColor; ctx.fillText("👾",0,8);
        // Shield
        if (bs.shieldHp>0) {
          ctx.strokeStyle="#60A5FA"; ctx.lineWidth=2; ctx.globalAlpha=0.5+Math.sin(Date.now()*0.005)*0.3;
          ctx.beginPath(); ctx.arc(0,0,40,0,Math.PI*2); ctx.stroke();
        }
        ctx.restore();
        // Boss HP Bar
        const bw=W*0.7, bx=(W-bw)/2, by=20;
        ctx.fillStyle="#1F2937"; ctx.fillRect(bx,by,bw,12); ctx.fillRect(bx-1,by-1,bw+2,14);
        const bossColor2 = bs.phase===3?"#EF4444":bs.phase===2?"#F59E0B":"#A78BFA";
        ctx.fillStyle=bossColor2; ctx.fillRect(bx,by,bw*(bs.hp/bs.maxHp),12);
        ctx.fillStyle="#fff"; ctx.font="10px 'Courier New'"; ctx.textAlign="center"; ctx.fillText(`👾 BOSS ${bs.hp}/${bs.maxHp}`,W/2,by+9); ctx.textAlign="left";
      }

      // Draw player ship
      ctx.save();
      ctx.translate(p.x+p.w/2, p.y+p.h/2);
      if (p.hit>0) { ctx.globalAlpha=0.3+Math.sin(p.hit*0.5)*0.7; }
      const shipColor = player.equippedShip==="ship_gold"?"#F59E0B":player.equippedShip==="ship_plasma"?"#8B5CF6":player.equippedShip==="ship_shadow"?"#6B7280":player.equippedShip==="ship_nova"?"#EF4444":"#60A5FA";
      ctx.shadowColor=shipColor; ctx.shadowBlur=15;
      // Ship body
      ctx.fillStyle=shipColor;
      ctx.beginPath(); ctx.moveTo(0,-16); ctx.lineTo(-12,12); ctx.lineTo(-5,6); ctx.lineTo(0,10); ctx.lineTo(5,6); ctx.lineTo(12,12); ctx.closePath(); ctx.fill();
      // Cockpit
      ctx.fillStyle="#111"; ctx.beginPath(); ctx.arc(0,-2,6,0,Math.PI*2); ctx.fill();
      ctx.fillStyle=shipColor+"88"; ctx.beginPath(); ctx.arc(0,-2,4,0,Math.PI*2); ctx.fill();
      // Shield indicator
      if (p.shield>0) { ctx.strokeStyle="#60A5FA"; ctx.lineWidth=2; ctx.globalAlpha=0.6+Math.sin(Date.now()*0.01)*0.4; ctx.beginPath(); ctx.arc(0,0,22,0,Math.PI*2); ctx.stroke(); }
      // Rapid fire
      if (p.rapid>0) { ctx.strokeStyle="#F59E0B"; ctx.lineWidth=1; ctx.globalAlpha=0.4; ctx.beginPath(); ctx.arc(0,0,18,0,Math.PI*2); ctx.stroke(); }
      ctx.restore();

      // Draw player bullets
      for (const b of s.bullets) {
        ctx.fillStyle = b.color||"#60A5FA";
        ctx.shadowColor = b.color||"#60A5FA";
        ctx.shadowBlur = 8;
        ctx.fillRect(b.x-b.w/2, b.y, b.w, b.h);
      }

      // Draw enemy bullets
      for (const b of s.enemyBullets) {
        ctx.fillStyle=b.color; ctx.shadowColor=b.color; ctx.shadowBlur=6;
        ctx.beginPath(); ctx.arc(b.x, b.y, b.isBoss?4:3, 0, Math.PI*2); ctx.fill();
      }

      // HUD - Score & Lives only
      ctx.shadowBlur=0;
      ctx.fillStyle="rgba(0,0,0,0.6)"; ctx.fillRect(0,0,W,44);
      ctx.fillStyle="#FFFFFF"; ctx.font="bold 18px 'Courier New'";
      ctx.fillText(s.score.toLocaleString(), 16, 28);
      // Lives (dots)
      for (let i=0;i<Math.min(p.lives,10);i++) {
        ctx.fillStyle="#60A5FA"; ctx.beginPath(); ctx.arc(W-16-i*14, 22, 5, 0, Math.PI*2); ctx.fill();
      }
      // Combo
      if (s.combo>1) {
        ctx.fillStyle="#F59E0B"; ctx.font="bold 13px 'Courier New'";
        ctx.textAlign="center"; ctx.fillText(`x${s.combo}`, W/2, 30); ctx.textAlign="left";
      }
      // Timer for time attack
      if (isTimeAttack) {
        const remaining=Math.max(0,60-s.elapsed);
        ctx.fillStyle=remaining<10?"#EF4444":"#10B981"; ctx.font="bold 13px 'Courier New'";
        ctx.textAlign="right"; ctx.fillText(`⏱ ${remaining.toFixed(1)}s`,W-70,30); ctx.textAlign="left";
      }
      // FPS
      if (player.settings.showFPS) {
        ctx.fillStyle="#374151"; ctx.font="10px 'Courier New'"; ctx.fillText(`${fps} FPS`,W-55,H-8);
      }

      ctx.restore();
      animRef.current = requestAnimationFrame(gameLoop);
    }

    function endGame() {
      cancelAnimationFrame(animRef.current);
      const s = stateRef.current;
      onEnd({ score:s.score, kills:s.killCount, survived:s.elapsed, combo:s.maxCombo, level:s.level, perfect:s.perfect, timeMs:(Date.now()-s.startTime) });
    }

    animRef.current = requestAnimationFrame(gameLoop);

    function onKey(e, down) {
      keysRef.current[e.key] = down;
      if (e.key==="Escape" && down) stateRef.current.paused = !stateRef.current.paused;
      if (["ArrowUp","ArrowDown","ArrowLeft","ArrowRight"," "].includes(e.key)) e.preventDefault();
    }
    window.addEventListener("keydown", e=>onKey(e,true));
    window.addEventListener("keyup", e=>onKey(e,false));

    return () => {
      cancelAnimationFrame(animRef.current);
      window.removeEventListener("keydown", e=>onKey(e,true));
      window.removeEventListener("keyup", e=>onKey(e,false));
    };
  }, []);

  const handleTouch = (dir, val) => {
    touchRef.current[dir] = val;
  };

  return (
    <div style={{ width:"100%", height:"100vh", background:"#000", display:"flex", flexDirection:"column", alignItems:"center", position:"relative", overflow:"hidden" }}>
      <canvas ref={canvasRef} style={{ display:"block", touchAction:"none" }} />

      {/* Touch Controls */}
      <div style={{ position:"fixed", bottom:20, left:0, right:0, display:"flex", justifyContent:"space-between", padding:"0 20px", pointerEvents:"none", zIndex:10 }}>
        {/* D-Pad */}
        <div style={{ display:"grid", gridTemplateColumns:"40px 40px 40px", gridTemplateRows:"40px 40px 40px", gap:4, pointerEvents:"all" }}>
          {[
            [null,1,null], [1,null,1], [null,1,null]
          ].map((row,ri) => row.map((cell,ci) => {
            const dirs = [[null,"up",null],[" left",null,"right"],[null,"down",null]];
            const icons = [[null,"▲",null],[["◀",null,"▶"]][0],["◀",null,"▶"],[null,"▼",null]];
            const dpad = [["▲",null,null],["◀",null,"▶"],[null,"▼",null]];
            const dir = ri===0&&ci===1?"up":ri===1&&ci===0?"left":ri===1&&ci===2?"right":ri===2&&ci===1?"down":null;
            const icon = ri===0&&ci===1?"▲":ri===1&&ci===0?"◀":ri===1&&ci===2?"▶":ri===2&&ci===1?"▼":null;
            if (!dir) return <div key={`${ri}-${ci}`} style={{ width:40,height:40 }} />;
            return (
              <button key={`${ri}-${ci}`}
                onTouchStart={e=>{e.preventDefault();handleTouch(dir,true);}}
                onTouchEnd={e=>{e.preventDefault();handleTouch(dir,false);}}
                onMouseDown={()=>handleTouch(dir,true)}
                onMouseUp={()=>handleTouch(dir,false)}

// MISSIONS SCREEN
// ============================================================
function MissionsScreen({ player, setScreen, theme, updatePlayer, showNotif, audio }) {
  const getDailyMissions = () => {
    const today = new Date().toDateString();
    if (player.lastDaily === today && player.dailyMissions.length > 0) {
      return player.dailyMissions;
    }
    const missions = DAILY_MISSIONS_DATA.map(m => ({ ...m, progress: 0, completed: false }));
    updatePlayer({ dailyMissions: missions, lastDaily: today });
    return missions;
  };

  const missions = getDailyMissions();
  const completed = missions.filter(m=>m.completed).length;

  const claimMission = (m) => {
    if (!m.completed) { showNotif("Mission non complétée!", "error", "❌"); return; }
    updatePlayer(prev => ({
      ...prev,
      coins: prev.coins + m.reward,
      dailyMissions: prev.dailyMissions.map(dm => dm.id===m.id?{...dm,claimed:true}:dm),
    }));
    showNotif(`+${m.reward} coins!`, "success", "💰");
    audio.play("coin");
  };

  return (
    <div style={{ maxWidth:480,margin:"0 auto",minHeight:"100vh",padding:"16px 16px 80px" }}>
      <div style={{ display:"flex",alignItems:"center",gap:12,marginBottom:16 }}>
        <button className="btn-secondary" style={{ padding:"8px 14px" }} onClick={()=>setScreen(SCREENS.DASHBOARD)}>← Retour</button>
        <h2 style={{ margin:0,fontSize:20,color:"#fff" }}>📋 Missions</h2>
      </div>

      {/* Daily Progress */}
      <div style={{ background:"linear-gradient(135deg,#064E3B,#065F46)",borderRadius:12,padding:16,marginBottom:16 }}>
        <div style={{ display:"flex",justifyContent:"space-between",marginBottom:8 }}>
          <span style={{ color:"#fff",fontWeight:"bold" }}>Missions Quotidiennes</span>
          <span style={{ color:"#6EE7B7" }}>{completed}/{missions.length}</span>
        </div>
        <div className="progress-bar" style={{ height:6 }}>
          <div className="progress-fill" style={{ width:`${(completed/missions.length)*100}%`,height:"100%",background:"#10B981" }}/>
        </div>
        <div style={{ fontSize:11,color:"#6EE7B7",marginTop:6 }}>Reset dans: {24-new Date().getHours()}h {60-new Date().getMinutes()}min</div>
      </div>

      <div style={{ display:"flex",flexDirection:"column",gap:10 }}>
        {missions.map(m=>(
          <div key={m.id} style={{ background:"#111827",border:`1px solid ${m.completed?"#10B98144":"#1F2937"}`,borderRadius:12,padding:16 }}>
            <div style={{ display:"flex",alignItems:"flex-start",gap:12 }}>
              <div style={{ width:44,height:44,borderRadius:10,background:m.completed?"#064E3B":"#1F2937",border:`1px solid ${m.completed?"#10B981":"#374151"}`,display:"flex",alignItems:"center",justifyContent:"center",fontSize:22 }}>
                {m.icon}
              </div>
              <div style={{ flex:1 }}>
                <div style={{ fontSize:13,color:"#fff",fontWeight:"bold",marginBottom:2 }}>{m.title}</div>
                <div style={{ fontSize:11,color:"#6B7280",marginBottom:8 }}>{m.desc}</div>
                <div className="progress-bar" style={{ height:5,marginBottom:4 }}>
                  <div className="progress-fill" style={{ width:`${Math.min(100,(m.progress/m.target)*100)}%`,height:"100%",background:m.completed?"#10B981":"#3B82F6" }}/>
                </div>
                <div style={{ fontSize:10,color:"#6B7280" }}>{Math.min(m.progress,m.target)}/{m.target}</div>
              </div>
              <div style={{ display:"flex",flexDirection:"column",alignItems:"flex-end",gap:6 }}>
                <div style={{ fontSize:12,color:"#F59E0B",fontWeight:"bold" }}>+{m.reward}💰</div>
                {m.completed && !m.claimed && <button className="btn-primary" style={{ fontSize:11,padding:"6px 10px" }} onClick={()=>claimMission(m)}>Réclamer</button>}
                {m.claimed && <div style={{ fontSize:10,color:"#10B981" }}>✓ Réclamé</div>}
              </div>
            </div>
          </div>
        ))}
      </div>

      {/* Weekly missions teaser */}
      <div style={{ background:"#111827",border:"1px solid #374151",borderRadius:12,padding:16,marginTop:16,opacity:0.6 }}>
        <div style={{ fontSize:13,color:"#9CA3AF",fontWeight:"bold",marginBottom:4 }}>📅 Missions Hebdomadaires</div>
        <div style={{ fontSize:11,color:"#6B7280" }}>Complétez toutes les missions quotidiennes pour débloquer!</div>
      </div>
    </div>
  );
}
